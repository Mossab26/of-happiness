document.addEventListener("DOMContentLoaded", () => { // 1. ترتيب أقسام الصفحة الرئيسية const sectionOrder = [ "home", "about", "activities", "team", "gallery", "join" ]; const main = document.querySelector("#main-content"); if (!main) { console.error("Main content not found!"); return; } // 2. إعادة ترتيب الأقسام sectionOrder.forEach(id => { const section = document.getElementById(id); if (section) { main.appendChild(section); } else { console.warn(`Section #${id} not found`); } }); // 3. تحديث قوائم التنقل const navContainers = [ ".nav-menu", ".mobile-nav-list", ".footer-nav-list" ]; navContainers.forEach(selector => { const nav = document.querySelector(selector); if (!nav) return; sectionOrder.forEach(id => { const link = nav.querySelector(`a[href="#${id}"]`); if (link && link.parentElement) { nav.appendChild(link.parentElement); } }); }); // 4. تحديث الرابط النشط عند النقر document.querySelectorAll(".nav-link").forEach(link => { link.addEventListener("click", () => { document.querySelectorAll(".nav-link") .forEach(item => item.classList.remove("active")); link.classList.add("active"); }); }); console.log("Homepage sections reordered successfully!"); });
